   173	            : `/api/proxy?url=${encodeURIComponent(item.file_url)}`;
   174	          debug.log(`Caching file for content ${item.id} from ${fetchUrl}`)
   175	          const res = await fetch(fetchUrl)
   176	          if (!res.ok) {
   177	            const text = await res.text().catch(() => '')
   178	            throw new Error(text || `fetch failed with status ${res.status}`)
   179	          }
   180	          const lengthHeader = res.headers.get('Content-Length')
   181	          if (lengthHeader && Number(lengthHeader) > MAX_CACHE_BYTES) {
   182	            toast({
   183	              title: 'File too large to cache',
   184	              description: 'This file exceeds the 50MB offline cache limit.'
   185	            })
   186	            res.body?.cancel?.()
   187	            return null
   188	          }
   189	          const array = await res.arrayBuffer()
   190	          if (array.byteLength === 0) {
   191	            debug.error('Fetched file is empty, not caching')
   192	            return null
   193	          }
   194	          if (array.byteLength > MAX_CACHE_BYTES) {
   195	            toast({
   196	              title: 'File too large to cache',
   197	              description: 'This file exceeds the 50MB offline cache limit.'
   198	            })
   199	            return null
   200	          }
   201	          const mime = res.headers.get('Content-Type') || 'application/octet-stream'
   202	          await localforage.setItem(key, { mime, data: array })
   203	          const size = array.byteLength
   204	          let index = await getIndex()
   205	          index.push({ id: item.id, size, lastAccess: Date.now() })
   206	          index = await enforceQuota(index)
   207	          await saveIndex(index)
   208	          debug.log(`Successfully cached file for content ${item.id}, size: ${size} bytes`)
   209	        }
   210	        return null // Return null for cache operations (not URL operations)
   211	      } catch (err) {
   212	        debug.error(`Failed to cache file for ${item.id}:`, err)
   213	        throw err
   214	      }
   215	    })()
   216	    
   217	    ongoingOperations.set(operationKey, cachePromise)
   218	    
   219	    try {
   220	      await cachePromise
   221	    } finally {
   222	      ongoingOperations.delete(operationKey)
   223	    }
   224	  }
   225	}

   232	export async function getCachedFileUrl(id: string): Promise<string | null> {
   233	  // Check if operation is already in progress
   234	  const operationKey = `get-${id}`
   235	  if (ongoingOperations.has(operationKey)) {
   236	    debug.log(`Get operation already in progress for ${id}, waiting...`)
   237	    return ongoingOperations.get(operationKey)!
   238	  }
   239	  
   240	  const getPromise = (async () => {
   241	    try {
   242	      const userId = await getUserId()
   243	      const stored = await localforage.getItem<any>(getFileKey(userId, id))
   244	      if (!stored) {
   245	        debug.log(`No cached file found for content ${id}`)
   246	        return null
   247	      }
   248	      
   249	      let index = await getIndex()
   250	      const entry = index.find(e => e.id === id)
   251	      if (entry) {
   252	        entry.lastAccess = Date.now()
   253	        await saveIndex(index)
   254	      }
   255	      
   256	      const BlobCtor: typeof Blob = (typeof window !== 'undefined' && (window as any).Blob) || Blob
   257	      const dataArray = stored.data instanceof ArrayBuffer ? new Uint8Array(stored.data) : new Uint8Array(stored.data)
   258	      const blob = new BlobCtor([dataArray], { type: stored.mime })
   259	      
   260	      if (typeof URL.createObjectURL === 'function') {
   261	        const url = URL.createObjectURL(blob)
   262	        debug.log(`Created blob URL for content ${id}: ${url.substring(0, 50)}...`)
   263	        return url
   264	      }
   265	      
   266	      const base64 = encodeBase64(dataArray)
   267	      const dataUrl = `data:${stored.mime};base64,${base64}`
   268	      debug.log(`Created data URL for content ${id}: ${dataUrl.substring(0, 50)}...`)
   269	      return dataUrl
   270	    } catch (err) {
   271	      debug.error('Failed to load cached file for', id, ':', err)
   272	      return null
   273	    }
   274	  })()
   275	  
   276	  ongoingOperations.set(operationKey, getPromise)
   277	  
   278	  try {
   279	    return await getPromise
   280	  } finally {
   281	    ongoingOperations.delete(operationKey)
   282	  }
   283	}

   345	export async function preloadContent(items: any[], signal?: AbortSignal): Promise<void> {
   346	  if (!items || items.length === 0) return
   347	  
   348	  for (const item of items) {
   349	    // Check for abort signal
   350	    if (signal?.aborted) {
   351	      debug.log('Preload operation aborted')
   352	      return
   353	    }
   354	    
   355	    if (!item?.id || !item?.file_url) continue
   356	    
   357	    try {
   358	      // Check if already cached to avoid unnecessary work
   359	      const userId = await getUserId()
   360	      const cached = await localforage.getItem(getFileKey(userId, item.id))
   361	      if (cached) {
   362	        debug.log(`Content ${item.id} already cached, skipping preload`)
   363	        continue
   364	      }
   365	      
   366	      // Use the existing caching mechanism
   367	      await cacheFileForContent(item)
   368	      
   369	      // Small delay to prevent overwhelming the system
   370	      if (!signal?.aborted) {
   371	        await new Promise(resolve => setTimeout(resolve, 50))
   372	      }
   373	    } catch (error) {
   374	      // Don't let preload failures break the system
   375	      debug.warn(`Failed to preload content ${item.id}:`, error)
   376	    }
   377	  }
   378	}

   380	export async function warmCache(priority: 'high' | 'normal' | 'low' = 'normal'): Promise<void> {
   381	  try {
   382	    const content = await getCachedContent()
   383	    if (content.length === 0) return
   384	    
   385	    // Sort by priority: recently accessed first, then by creation date
   386	    const sorted = content.sort((a, b) => {
   387	      const aAccess = a.lastAccess || a.created_at || 0
   388	      const bAccess = b.lastAccess || b.created_at || 0
   389	      return new Date(bAccess).getTime() - new Date(aAccess).getTime()
   390	    })
   391	    
   392	    // Limit based on priority level
   393	    const limit = priority === 'high' ? 5 : priority === 'normal' ? 3 : 1
   394	    const toWarm = sorted.slice(0, limit)
   395	    
   396	    debug.log(`Warming cache for ${toWarm.length} items (priority: ${priority})`)
   397	    
   398	    // Warm cache in background without waiting
   399	    cacheFilesForContent(toWarm).catch(error => {
   400	      debug.warn('Cache warming failed:', error)
   401	    })
   402	  } catch (error) {
   403	    debug.warn('Failed to warm cache:', error)
   404	  }
   405	}

   415	export async function clearOfflineContent(userIdArg?: string): Promise<void> {
   416	  const userId = userIdArg ?? (await getUserId())
   417	  const indexKey = getIndexKey(userId)
   418	  const storeKey = getStoreKey(userId)
   419	  const index = (await localforage.getItem<IndexEntry[]>(indexKey)) || []
   420	  for (const entry of index) {
   421	    await localforage.removeItem(getFileKey(userId, entry.id))
   422	  }
   423	  await localforage.removeItem(indexKey)
   424	  await localforage.removeItem(storeKey)
   425	  
   426	  // Reset cache metrics on clear
   427	  cacheMetrics.hitCount = 0
   428	  cacheMetrics.missCount = 0
   429	  cacheMetrics.hitRate = 0
   430	  cacheMetrics.lastOptimization = Date.now()
   431	}
