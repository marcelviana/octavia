    76	function addTokenToUserSession(userId: string, token: string): void {
    77	  if (!userSessionMap.has(userId)) {
    78	    userSessionMap.set(userId, new Set())
    79	  }

    83	function removeTokenFromUserSession(userId: string, token: string): void {
    84	  const userTokens = userSessionMap.get(userId)
    85	  if (userTokens) {
    86	    userTokens.delete(token)
    87	    if (userTokens.size === 0) {
    88	      userSessionMap.delete(userId)
    89	    }
    90	  }

    94	 * SECURITY: Immediately blacklist a token (for logout, user suspension, etc.)
    95	 */
    96	export function blacklistToken(token: string): void {
    97	  if (!token) return
    98	
    99	  const exp = Date.now() + SECURITY_CONFIG.BLACKLIST_DURATION_MS
   100	  tokenBlacklist.set(token, exp)
   101	
   102	  // Remove from cache immediately
   103	  const cachedEntry = tokenCache.get(token)
   104	  if (cachedEntry) {
   105	    removeTokenFromUserSession(cachedEntry.userId, token)
   106	    tokenCache.delete(token)
   107	  }
   108	
   109	  logger.log(`Token blacklisted: ${token.substring(0, 10)}...`)
   110	}

   115	export function blacklistUserTokens(userId: string): void {

   128	function isTokenBlacklisted(token: string): boolean {
   129	  const exp = tokenBlacklist.get(token)
   130	  if (!exp) return false
   131	
   132	  if (Date.now() > exp) {
   133	    tokenBlacklist.delete(token)
   134	    return false
   135	  }
   136	
   137	  return true
   138	}

   140	/**
   141	 * SECURITY-HARDENED: Validate Firebase token with enhanced security checks
   142	 */
   143	export async function validateFirebaseTokenSecure(
   144	  idToken: string,
   145	  requestUrl?: string | URL
   146	): Promise<ServerAuthResult> {
   147	  try {
   148	    if (!idToken || typeof idToken !== 'string') {
   149	      return { isValid: false, error: 'Missing or invalid ID token format' }
   150	    }
   151	
   152	    // SECURITY: Check token blacklist first
   153	    if (isTokenBlacklisted(idToken)) {
   154	      logger.warn('Attempted use of blacklisted token')
   155	      return { isValid: false, error: 'Token has been revoked' }
   156	    }
   157	
   158	    // Check cache with security considerations
   159	    const cached = tokenCache.get(idToken)
   160	    const now = Date.now()
   161	
   162	    if (cached && cached.exp > now) {
   163	      logger.log('Token validation from cache (secure)')
   164	      return cached.result
   165	    }
   166	
   167	    // Remove expired cached entry
   168	    if (cached && cached.exp <= now) {
   169	      removeTokenFromUserSession(cached.userId, idToken)
   170	      tokenCache.delete(idToken)
   171	    }
   172	
   173	    // Validate via API with enhanced error handling
   174	    let result: ServerAuthResult
   175	
   176	    if (typeof window === 'undefined' && process.env.NEXT_RUNTIME !== 'edge') {
   177	      // B1.1: chamada direta de função nas lambdas Node — mesmo guard
   178	      // estático da cadeia A (lib/firebase-server-utils.ts); ver o
   179	      // comentário lá sobre DCE por compilação e o alias do client.
   180	      try {
   181	        const { verifyFirebaseToken } = await import('@/lib/firebase-admin')
   182	        const decoded = await verifyFirebaseToken(idToken)
   183	
   184	        result = {
   185	          isValid: true,
   186	          user: {
   187	            uid: decoded.uid,
   188	            email: decoded.email,
   189	            emailVerified: decoded.email_verified
   190	          }
   191	        }
   192	
   193	        // SECURITY: Track user session
   194	        addTokenToUserSession(decoded.uid, idToken)
   195	
   196	        logger.log(`Token validated successfully for user: ${decoded.uid}`)
   197	      } catch (error: any) {
   198	        const msg = String(error?.message || '')
   199	        // Mesmas classes observáveis do transporte HTTP (o auto-blacklist
   200	        // por string de erro segue dormante, como hoje — nota do desenho
   201	        // da B1.1; acordá-lo é decisão do B1.5):
   202	        // token inválido/expirado → 'Token verification failed' (era o
   203	        // !response.ok); erro de infra → 'service unavailable' (era o
   204	        // catch do fetch).
   205	        if (/expired|invalid|argument|decod/i.test(msg)) {
   206	          logger.error(`Token verification failed: token inválido/expirado`)
   207	          return { isValid: false, error: 'Token verification failed' }
   208	        }
   209	        logger.error('Token verification API call failed:', error)
   210	        return { isValid: false, error: 'Token verification service unavailable' }
   211	      }
   212	    } else if (typeof window === 'undefined') {
   213	      // B1.2b: o ramo fetch (self-fetch a /api/auth/verify, âncora de env
   214	      // da PR #221) morreu junto com a rota — sem consumidor em produção
   215	      // desde a B1.1. A verificação só existe no runtime Node (guard acima).
   216	      logger.error('Token validation requested outside Node runtime')
   217	      return { isValid: false, error: 'Token verification failed' }
   218	    } else {
   219	      // Client-side or fallback
   220	      logger.warn('Client-side token validation attempted - security risk')
   221	      return { isValid: false, error: 'Client-side validation not allowed' }
   222	    }
   223	
   224	    // Cache with security considerations
   225	    if (result.isValid && result.user) {
   226	      tokenCache.set(idToken, {
   227	        result,
   228	        exp: now + SECURITY_CONFIG.TOKEN_CACHE_DURATION_MS,
   229	        userId: result.user.uid
   230	      })
   231	    }
   232	
   233	    return result
   234	
   235	  } catch (error) {
   236	    logger.error('Token validation error:', error)
   237	    return { isValid: false, error: 'Token validation failed' }
   238	  }
   239	}

   241	/**
   242	 * SECURITY-HARDENED: Require authentication with enhanced security
   243	 */
   244	export async function requireAuthServerSecure(
   245	  request: Request,
   246	  options: { allowUnverifiedEmail?: boolean } = {}
   247	): Promise<{
   248	  uid: string
   249	  email?: string
   250	  emailVerified?: boolean
   251	} | null> {
   252	  try {
   253	    // B1.3: fallback por IP para auth falhada — mesmo padrão do funil A
   254	    // (lib/firebase-server-utils.ts): IP estourado nega sem verificar.
   255	    const clientIp = getClientIp(request)
   256	    if (authFailureLimited(clientIp)) {
   257	      return null
   258	    }
   259	
   260	    // Extract token with security validation
   261	    const authHeader = request.headers.get('authorization')
   262	    const cookieHeader = request.headers.get('cookie')
   263	
   264	    let token: string | null = null
   265	
   266	    // Extract from Authorization header
   267	    if (authHeader) {
   268	      const bearerMatch = authHeader.match(/^Bearer\s+(.+)$/i)
   269	      if (bearerMatch && bearerMatch[1]) {
   270	        token = bearerMatch[1]
   271	      } else {
   272	        logger.warn('Invalid Authorization header format')
   273	        return null
   274	      }
   275	    }
   276	    // Extract from session cookie
   277	    else if (cookieHeader) {
   278	      const sessionMatch = cookieHeader.match(/firebase-session=([^;]+)/)
   279	      if (sessionMatch && sessionMatch[1]) {
   280	        token = decodeURIComponent(sessionMatch[1])
   281	      }
   282	    }
   283	
   284	    if (!token) {
   285	      logger.log('No authentication token found')
   286	      return null
   287	    }
   288	
   289	    // Validate token with security enhancements
   290	    const validation = await validateFirebaseTokenSecure(token, request.url)
   291	
   292	    if (!validation.isValid || !validation.user) {
   293	      logger.warn('Token validation failed during auth requirement')
   294	      recordAuthFailure(clientIp)
   295	
   296	      // SECURITY: Automatically blacklist suspicious tokens
   297	      if (validation.error?.includes('revoked') || validation.error?.includes('expired')) {
   298	        blacklistToken(token)
   299	      }
   300	
   301	      return null
   302	    }
   303	
   304	    // SECURITY: Additional email verification check.
   305	    // allowUnverifiedEmail existe para fluxos onde o usuário acabou de ser
   306	    // criado e ainda não pôde verificar o email (ex.: criação de perfil no signup)
   307	    if (!validation.user.emailVerified && !options.allowUnverifiedEmail) {
   308	      logger.warn(`Unverified user attempted access: ${validation.user.uid}`)
   309	      return null
   310	    }
   311	
   312	    logger.log(`Authentication successful for user: ${validation.user.uid}`)
   313	    return validation.user
   314	
   315	  } catch (error) {
   316	    logger.error('Authentication requirement failed:', error)
   317	    return null
   318	  }
   319	}
