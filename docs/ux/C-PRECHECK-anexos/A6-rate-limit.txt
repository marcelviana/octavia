    93	export function checkRateLimit(opts: {
    94	  scope: 'user' | 'ip'
    95	  id: string
    96	  familia: string
    97	  config: RateLimitConfig
    98	}): RateLimitResult {
    99	  const { scope, id, familia, config } = opts
   100	  const key = `${scope}:${id}:${familia}`
   101	  const now = Date.now()
   102	
   103	  let entry = store.get(key)
   104	  if (!entry || now > entry.resetTime) {
   105	    entry = { count: 0, resetTime: now + config.windowMs }
   106	    store.set(key, entry)
   107	  }
   108	  entry.count++
   109	
   110	  return {
   111	    ok: entry.count <= config.max,
   112	    scope,
   113	    limit: config.max,
   114	    remaining: Math.max(0, config.max - entry.count),
   115	    resetTime: entry.resetTime,
   116	  }
   117	}

   119	/**
   120	 * Resposta 429 estruturada com a assinatura do sistema único.
   121	 * B3 PR-4/D4: corpo pelo ponto único — ganha code:'RATE_LIMITED';
   122	 * retryAfter e os 5 headers ficam como estão. Direção de import
   123	 * user-rate-limit → api-errors (api-errors é Response puro, sem ciclo).
   124	 */
   125	export function rateLimited(result: RateLimitResult): Response {
   126	  const retryAfter = Math.max(1, Math.ceil((result.resetTime - Date.now()) / 1000))
   127	  return apiError('RATE_LIMITED', 'Rate limit exceeded', {
   128	    extra: { retryAfter },
   129	    headers: {
   130	      'X-RateLimit-Limit': String(result.limit),
   131	      'X-RateLimit-Remaining': String(result.remaining),
   132	      'X-RateLimit-Reset': String(result.resetTime),
   133	      'X-RateLimit-Scope': result.scope,
   134	      'Retry-After': String(retryAfter),
   135	    },
   136	  })
   137	}
