import { chromium } from '@playwright/test'
async function main() {
  const b = await chromium.launch({ channel: 'chrome', headless: true })
  for (const comRoute of [true, false]) {
    const ctx = await b.newContext()
    if (comRoute) await ctx.route('**/*', (r) => r.continue())
    const page = await ctx.newPage()
    const antigo: string[] = [], novoReq: string[] = [], novoResp: string[] = [], fin: string[] = []
    page.on('requestfinished', async (r) => { const st = String((await r.response())?.status() ?? '-'); if (r.url().includes('cn=')) antigo.push(`${new URL(r.url()).search} ${st}`) })
    ctx.on('request', (r) => { if (r.url().includes('cn=')) novoReq.push(new URL(r.url()).search) })
    ctx.on('response', (r) => { if (r.url().includes('cn=')) novoResp.push(`${new URL(r.url()).search} ${r.status()}`) })
    ctx.on('requestfinished', (r) => { if (r.url().includes('cn=')) fin.push(new URL(r.url()).search) })
    await page.goto('http://localhost:3000/login', { waitUntil: 'domcontentloaded' })
    await page.evaluate(() => fetch('/api/health?cn=1').then((r) => r.status))
    await page.evaluate(() => fetch('/api/health?cn=2').then((r) => r.json()))
    await page.evaluate(() => { void fetch('/api/health?cn=3'); window.location.href = '/login?cn=nav' })
    await page.waitForURL('**/login?cn=nav'); await page.waitForTimeout(1500)
    console.log(JSON.stringify({ comRoute, antigo_page_requestfinished: antigo, novo_ctx_request: novoReq, novo_ctx_response: novoResp, ctx_requestfinished: fin }))
    await ctx.close()
  }
  await b.close()
}
