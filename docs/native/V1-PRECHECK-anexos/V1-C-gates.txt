#!/bin/sh
# G2 — testIDs: o conjunto ANTES é subconjunto do DEPOIS (nenhum some).
# G3 — linhas `log(` idênticas por arquivo (os logs são o instrumento dos aceites).
# Uso: g2g3.sh <ref-antes> <ref-depois-ou-WORKTREE>
A=$1; B=$2
tmp=$(mktemp -d)
coleta() { # $1=ref  $2=saida-testids  $3=saida-logs
  if [ "$1" = "WORKTREE" ]; then
    for f in $(find apps/native/src -name '*.ts' -o -name '*.tsx' | sort); do
      grep -o 'testID="[^"]*"' "$f" 2>/dev/null | sed "s|^|$f\t|"
      grep -oE 'testID=\{`[^`]*`\}' "$f" 2>/dev/null | sed "s|^|$f\t|"
    done | sort -u > "$2"
    for f in $(find apps/native/src packages/core/src -name '*.ts' -o -name '*.tsx' | sort); do
      grep -n 'log(' "$f" 2>/dev/null | sed 's/^[0-9]*://' | sed 's/^ *//' | sed "s|^|$f\t|"
    done | sort > "$3"
  else
    for f in $(git ls-tree -r --name-only "$1" -- apps/native/src | sort); do
      git show "$1:$f" 2>/dev/null | grep -o 'testID="[^"]*"' | sed "s|^|$f\t|"
      git show "$1:$f" 2>/dev/null | grep -oE 'testID=\{`[^`]*`\}' | sed "s|^|$f\t|"
    done | sort -u > "$2"
    for f in $(git ls-tree -r --name-only "$1" -- apps/native/src packages/core/src | sort); do
      git show "$1:$f" 2>/dev/null | grep 'log(' | sed 's/^ *//' | sed "s|^|$f\t|"
    done | sort > "$3"
  fi

else echo "  G3: DIVERGEM ✗"; echo "$DIF" | head -12 | sed 's/^/      /'; G3=1; fi
rm -rf $tmp
[ $G2 -eq 0 ] && [ $G3 -eq 0 ]
