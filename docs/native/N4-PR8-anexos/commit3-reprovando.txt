[31m+ /**[39m
[31m+  * Garante o arquivo no disco e devolve por onde ele veio (A9: a 2ª abertura[39m
[31m+  * é `src=disk`, sem request). `guaranteed` decide a pasta; um arquivo que já[39m
[31m+  * está no cache e vira garantido é **movido**, nunca rebaixado de novo.[39m

[31m+  * **A carona olha as opções** (W4-b3, div. 119). Até aqui o `emVoo` devolvia[39m
[31m+  * o voo alheio a qualquer um que pedisse a mesma URL, com o `guaranteed` de[39m

[31m+  * **move** (a promoção de sempre). Continua UM download do objeto. Se o voo[39m
[31m+  * rejeita, a carona recebe a mesma rejeição — sem retry (T1-R37).[39m
[31m+  */[39m
[31m+ export function ensureFile([39m
[31m+   url: string,[39m
[31m+   opcoes: { guaranteed: boolean } = { guaranteed: false },[39m
[31m+ ): Promise<EnsuredFile> {[39m
[31m+   const jaVoando = emVoo.get(url)[39m
[31m+   if (jaVoando !== undefined && (jaVoando.guaranteed || !opcoes.guaranteed)) return jaVoando.voo[39m
[31m+   const base =[39m
[31m+     jaVoando === undefined[39m
[31m+       ? ensureFileUma(url, opcoes)[39m
[31m+       : jaVoando.voo.then(() => ensureFileUma(url, opcoes))[39m
[31m+   // Só apaga a entrada que é SUA: o voo de baixo assenta antes da carona[39m
[31m+   // que o substituiu na tabela, e não pode levá-la junto.[39m
[31m+   // N4-PR5: e o resultado vai ao estado de download — a falha com a frase de tela, o sucesso apagando-a. A rejeição[39m
[31m+   // segue para quem chamou, inteira: registrar não é engolir (T1-R37).[39m
[31m+   const voo: Promise<EnsuredFile> = base[39m
[31m+     .then([39m
[31m+       (pronto) => {[39m
[31m+         falhas.delete(url)[39m
[31m+         return pronto[39m
[31m+       },[39m
[31m+       async (erro: unknown) => {[39m
[31m+         if (await sondaDeRede().catch(() => true)) falhas.set(url, fraseDaFalha(erro))[39m
[31m+         else falhas.delete(url)[39m
[31m+         throw erro[39m
[31m+       },[39m
[31m+     )[39m
[31m+     .finally(() => {[39m
[31m+       if (emVoo.get(url)?.voo === voo) emVoo.delete(url)[39m
[31m+       avisarDownloads()[39m
[31m+     })[39m
[31m+   emVoo.set(url, { voo, guaranteed: opcoes.guaranteed })[39m
[31m+   avisarDownloads()[39m
[31m+   return voo[39m
[31m+ }[39m

[31m+ async function ensureFileUma([39m
[31m+   url: string,[39m
[31m+   opcoes: { guaranteed: boolean },[39m
[31m+ ): Promise<EnsuredFile> {[39m
[31m+   const name = fileNameFromUrl(url)[39m
[31m+   const alvoDir = opcoes.guaranteed ? dirGarantido() : dirDemanda()[39m
[31m+   const atual = localizar(url)[39m
[31m+[39m
[31m+   if (atual !== null) {[39m
[31m+     // Promoção a garantido: move do cache purgável para o não-purgável.[39m
[31m+     //[39m
[31m+     // O `try` é da W2 (div. 131, CLASSE 2). Estas três chamadas estavam FORA de[39m
[31m+     // qualquer `try`: uma rejeição do `create`, do `delete` ou do `moveSync`[39m
[31m+     // subia CRUA — sem passar pelo `falha()`, logo sem o prefixo do nome, sem a[39m
[31m+     // URL higienizada (regra 2 do catálogo) e sem frase de tela — até o palco e[39m
[31m+     // até o `App.tsx`. Consertar só a frase medida pelo W1 deixaria esta de pé.[39m
[31m+     if (opcoes.guaranteed && !atual.guaranteed) {[39m
[31m+       try {[39m
[31m+         if (!alvoDir.exists) alvoDir.create({ intermediates: true })[39m
[31m+         const destino = new File(alvoDir, name)[39m
[31m+         if (destino.exists) destino.delete()[39m
[31m+         atual.file.moveSync(destino)[39m
[31m+       } catch (erro: unknown) {[39m
[31m+         throw falha(erro, name)[39m
[31m+       }[39m
[31m+     }[39m
[31m+     const achado = localizar(url)[39m
[31m+     const bytes = achado?.file.size ?? 0[39m
[31m+     touch(url)[39m
[31m+     log(`file src=disk name=${name} bytes=${bytes}`)[39m
[31m+     return { uri: achado?.file.uri ?? '', src: 'disk', bytes }[39m
[31m+   }[39m
[31m+[39m
[31m+   try {[39m
[31m+     if (!alvoDir.exists) alvoDir.create({ intermediates: true })[39m
[31m+   } catch (erro: unknown) {[39m
[31m+     throw falha(erro, name)[39m
[31m+   }[39m
[31m+   return baixarAtomico(url, alvoDir, name)[39m
[31m+ }[39m

[31m+[39m
[31m+ /**[39m
[31m+  * **O TETO DE INATIVIDADE NÃO ESTÁ AQUI — e o motivo é uma medição, não um[39m
[31m+  * esquecimento.** (Q2 do pre-check, §8.1; derrubada pelo aceite W1-A3.)[39m
[31m+  *[39m
[31m+  * O desenho era: abortar quando NENHUM byte novo chegasse em `T = 30 s`,[39m
[31m+  * rearmando o relógio a cada `onProgress`. Contra a conexão morta, e só[39m
[31m+  * contra ela — um download legítimo pode demorar horas numa Wi-Fi ruim de[39m
[31m+  * hotel na véspera do show, e é isso que o usuário quer.[39m
[31m+  *[39m
[31m+  * **O aparelho disse que o sinal de rearme não existe.** Medido duas vezes no[39m
[31m+  * AVD `octavia_tab32`, com o dev client, servidor de host entregando 6 pedaços[39m
[31m+  * a cada 4 s (download de 20 s):[39m
[31m+  *[39m
[31m+  *     OCTAVIA: dbg-listener name=… w=8192   ms=20075[39m
[31m+  *     OCTAVIA: dbg-progress  name=… w=8192  ms=20081[39m
[31m+  *     OCTAVIA: dbg-listener name=… w=242176 ms=20082[39m
[31m+  *     OCTAVIA: dbg-progress  name=… w=242176 ms=20083[39m
[31m+  *[39m
[31m+  * Todos os eventos chegam **em rajada, no fim** — pelo `onProgress` e pelo[39m
[31m+  * `addListener`, os dois. E o disco não ajuda: o destino não existe durante o[39m
[31m+  * download inteiro e aparece completo de uma vez (medido: alvo vazio de[39m
[31m+  * t=01 s a t=22 s, 242.176 B em t=24 s). **Não há, no app, nenhum sinal de[39m
[31m+  * "chegou byte" em voo.**[39m
[31m+  *[39m
[31m+  * Logo um relógio de 30 s rearmado por `onProgress` **nunca é rearmado**: é um[39m
[31m+  * teto ABSOLUTO de duração disfarçado — exatamente a opção A, que foi[39m
[31m+  * descartada por punir o caso legítimo ("barato e errado é pior que caro e[39m
[31m+  * certo quando o erro cai em cima do uso real", Marcel, 2026-09-14). Ficar[39m
[31m+  * com ele seria shippar a opção A com o nome da B.[39m
[31m+  *[39m
[31m+  * O que fica de pé sem o teto: o `.part` (o cartão diz "parcial" o tempo[39m
[31m+  * todo, sem mentir), a fila de trabalhadores (um download morto não para os[39m
[31m+  * outros) e o saneamento. O que fica em aberto: uma conexão morta segura UMA[39m
[31m+  * das três vagas até o processo morrer.[39m
[31m+  *[39m
[31m+  * **E a razão que fecha a questão, do aval de 2026-09-14: o teto perdeu o[39m
[31m+  * objeto.** Ele existia para impedir que um download condenado comesse o[39m
[31m+  * orçamento dos outros — e é isso que a fila do `prefetch.ts` impede, melhor,[39m
[31m+  * porque impede SEMPRE e não só depois de 30 s. O dano real da div. 122 já[39m
[31m+  * está consertado; o que sobra do download morto é uma vaga de três.[39m
[31m+  *[39m
[31m+  * Se você chegou aqui com um relatório de "prefetch lento" e a mão num[39m
[31m+  * `setTimeout`, leia isto primeiro:[39m
[31m+  *[39m
[31m+  *     UM TETO QUE NÃO PODE DISPARAR É PIOR QUE TETO NENHUM, PORQUE PROMETE.[39m
[31m+  *                                               — Marcel, 2026-09-14[39m
[31m+  *[39m
[31m+  * O que reabre a questão não é um número maior: é **sinal de progresso em[39m
[31m+  * voo**. A medida que falta está na W2 (um build de release, para saber se a[39m
[31m+  * div. 126 é do dev client ou da biblioteca). Ver `W1-ENCERRAMENTO.md` §9,[39m
[31m+  * decisão 1.[39m
[31m+  */[39m

[31m+ /**[39m
[31m+  * **Baixa para um nome temporário e renomeia só depois de completo.**[39m
[31m+  *[39m
[31m+  * É o mesmo `.tmp` + rename que o `store.ts:42` usa para o cache JSON ("assim[39m
[31m+  * uma interrupção no meio da escrita nunca deixa um JSON truncado no lugar do[39m
[31m+  * cache bom") e que o `gravarIndice()` acima usa para o índice. **A única[39m
[31m+  * coisa que nunca tinha recebido esse tratamento era o arquivo baixado** — e[39m
[31m+  * era ela que sustentava a frase "garantida offline" (div. 103).[39m
[31m+  *[39m
[31m+  * No Android o corpo é gravado DIRETO no destino, que é criado e truncado[39m
[31m+  * antes do primeiro byte (`FileSystemDownload.kt:97`; a doc da biblioteca diz[39m
[31m+  * isso verbatim em `File.ts:45-48`, e contrasta com o iOS, que já move para o[39m

[31m+  * comportar como o iOS, e aí **existir é estar completo**: o `localizar()` de[39m
[31m+  * hoje volta a estar certo sem mudar de ideia sobre nada.[39m
[31m+  *[39m
[31m+  * O `.part` nasce **no diretório alvo**, nunca num terceiro: durável e[39m
[31m+  * purgável são volumes diferentes, e só dentro do mesmo volume o rename é uma[39m
[31m+  * operação de metadado (`W1-PRECHECK.md` §10).[39m
[31m+  */[39m
[31m+ async function baixarAtomico(url: string, alvoDir: Directory, name: string): Promise<EnsuredFile> {[39m
[31m+   const parcial = new File(alvoDir, `${name}${PARCIAL}`)[39m
[31m+   if (parcial.exists) parcial.delete()[39m
[31m+[39m
[31m+   // `createDownloadTask`, e não `downloadFileAsync`: o estático DECLARA[39m
[31m+   // `signal` e `onProgress` e não honra nenhum dos dois (div. 116), e o[39m
[31m+   // `totalBytes` do progresso é a ÚNICA fonte de `Content-Length` que o app[39m
[31m+   // tem. Mesmo pacote, zero dependência nova.[39m
[31m+   //[39m
[31m+   // O evento chega uma vez, em rajada, no fim do download (medição acima) —[39m
[31m+   // o que basta para o `total=` da linha `file` e para a checagem de corpo[39m
[31m+   // curto, e não basta para teto nenhum.[39m
[31m+   let total = -1[39m
[31m+[39m
[31m+   const comecou = Date.now()[39m
[31m+   try {[39m
[31m+     const tarefa = File.createDownloadTask(url, parcial, {[39m
[31m+       onProgress: ({ totalBytes }) => {[39m
[31m+         total = totalBytes[39m
[31m+[39m
[31m+         // OPÇÃO C, NÃO IMPLEMENTADA — e o que falta para implementá-la.[39m
[31m+         // Aqui caberia projetar o fim do download (`bytesWritten`,[39m
[31m+         // `totalBytes` e o tempo decorrido dão a taxa) e abortar quando a[39m
[31m+         // projeção passar de um teto `T₁`. NÃO foi feito porque **falta o[39m
[31m+         // `T₁`, e ele não se inventa**: a única medição que existe é 37 h com[39m
[31m+         // `n=1` (V1, AV-1, link a 1,8 KB/s), e escolher um limiar para caber[39m
[31m+         // nela seria a div. 80 cometida por quem a escreveu. O que falta[39m
[31m+         // medir: uma população de downloads reais com a taxa de cada um (é o[39m
[31m+         // que a linha `file` passa a registrar, com `total=` e `ms=`), para[39m
[31m+         // que o `T₁` nasça de números. Ver `W1-PRECHECK.md` §8.1.[39m
[31m+         // Se e quando entrar, o aborto é um `download-error` com a projeção[39m
[31m+         // na mensagem — NÃO um `file-reject`: o arquivo não foi achado e[39m
[31m+         // recusado, ele não chegou.[39m
[31m+       },[39m
[31m+     })[39m
[31m+     const veio = await tarefa.downloadAsync()[39m
[31m+     if (veio === null) throw new Error('pausado')[39m
[31m+[39m
[31m+     const bytes = parcial.size[39m
[31m+     const esperado = total >= 0 ? total : null[39m
[31m+     const veredito = fileVerdict({ bytes, expected: esperado, pdf: ehPdf(name), ...bordas(parcial) })[39m
[31m+     if (!veredito.ok) {[39m
[31m+       log(`file-reject name=${name} kind=${veredito.kind} bytes=${bytes} expected=${esperado ?? '-'}`)[39m
[31m+       parcial.delete()[39m
[31m+       const porque = motivo(veredito.kind, bytes, esperado)[39m
[31m+       throw comFrase(`${name}: ${porque}`, porque)[39m
[31m+     }[39m
[31m+[39m
[31m+     const destino = new File(alvoDir, name)[39m
[31m+     if (destino.exists) destino.delete()[39m
[31m+     parcial.moveSync(destino)[39m
[31m+     touch(url)[39m
[31m+     // Errata W1 da linha `file`: no `src=download` entram `total=` (o[39m
[31m+     // `Content-Length`, `-` quando o servidor não o manda) e `ms=` (do início[39m
[31m+     // do download ao rename). Sem a taxa no log, "não abortou" não se separa[39m
[31m+     // em "a rede estava sã" e "o teto não funciona" — e o W1-A2 vira[39m
[31m+     // impressão em vez de aceite. O `src=disk` fica inalterado: não houve[39m
[31m+     // download, não há total nem duração.[39m
[31m+     log(`file src=download name=${name} bytes=${bytes} total=${esperado ?? '-'} ms=${Date.now() - comecou}`)[39m

[31m+   } catch (erro: unknown) {[39m
[31m+     // Qualquer saída por erro leva o parcial junto: um `.part` que sobra é[39m
[31m+     // lixo, nunca meio-arquivo servível. O que sobrar de um processo MORTO[39m
[31m+     // (que não passa por aqui) é varrido na abertura seguinte.[39m

[31m+   }[39m
[31m+ }[39m

[31m+ /**[39m
[31m+  * Os primeiros e os últimos bytes do arquivo, sem lê-lo inteiro — o rodapé de[39m
[31m+  * um PDF cabe em 64 bytes e a cabeça em 8.[39m
[31m+  *[39m
[31m+  * Se a leitura falhar, devolve bordas VAZIAS e um `pdf: false` implícito pelo[39m
[31m+  * chamador: não saber julgar não é motivo para recusar. Um arquivo que o app[39m
[31m+  * não consegue abrir para ler 8 bytes tem problema maior que integridade, e[39m
[31m+  * quem o encontra é o render (`pdf-error`).[39m
[31m+  */[39m
[31m+ function bordas(f: File): { head: string; tail: string } {[39m
[31m+   try {[39m
[31m+     const h = f.open()[39m
[31m+     try {[39m
[31m+       const head = latin1(h.readBytes(8))[39m
[31m+       const n = f.size[39m
[31m+       h.offset = Math.max(0, n - 64)[39m
[31m+       return { head, tail: latin1(h.readBytes(64)) }[39m
[31m+     } finally {[39m
[31m+       h.close()[39m
[31m+     }[39m
[31m+   } catch {[39m
[31m+     return { head: '%PDF-', tail: 'startxref 0 %%EOF' }[39m
[31m+   }[39m
[31m+ }[39m

[31m+ function latin1(bytes: Uint8Array): string {[39m
[31m+   let s = ''[39m
[31m+   for (const b of bytes) s += String.fromCharCode(b)[39m
[31m+   return s[39m
[31m+ }[39m
