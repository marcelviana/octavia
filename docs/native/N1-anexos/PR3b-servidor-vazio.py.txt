# Prova de S1a (sincronizando, sem cache) e S1f (vazia após sync ok), sem
# tocar em prod: /api/setlists demora 9 s e devolve []; /api/content devolve
# a página vazia no shape real do handler.
from http.server import BaseHTTPRequestHandler, HTTPServer
import time, json
class H(BaseHTTPRequestHandler):
    def do_GET(self):
        if self.path.startswith('/api/setlists'):
            time.sleep(9)
            corpo = json.dumps([]).encode()
        else:
            corpo = json.dumps({'data': [], 'total': 0, 'page': 1, 'pageSize': 100,
                                'hasMore': False, 'totalPages': 0}).encode()
        self.send_response(200)
        self.send_header('Content-Type', 'application/json')
        self.send_header('Cache-Control', 'private, no-store')
        self.send_header('Content-Length', str(len(corpo)))
        self.end_headers()
        self.wfile.write(corpo)
    def log_message(self, *a):
        print('REQ', self.path, flush=True)
HTTPServer(('127.0.0.1', 8099), H).serve_forever()
