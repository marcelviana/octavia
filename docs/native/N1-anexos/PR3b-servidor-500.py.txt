# Servidor de prova do S1e/A21: responde 500 com o envelope do CONTRATO-DE-ERRO
# em qualquer rota. Não toca em prod; roda só no host, alcançado por adb reverse.
from http.server import BaseHTTPRequestHandler, HTTPServer
CORPO = b'{"error":"Internal server error","code":"INTERNAL_ERROR"}'
class H(BaseHTTPRequestHandler):
    def do_GET(self):
        self.send_response(500)
        self.send_header('Content-Type', 'application/json')
        self.send_header('Content-Length', str(len(CORPO)))
        self.end_headers()
        self.wfile.write(CORPO)
    def log_message(self, *a): 
        print('REQ', self.path, flush=True)
HTTPServer(('127.0.0.1', 8099), H).serve_forever()
