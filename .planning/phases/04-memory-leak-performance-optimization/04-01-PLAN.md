Consolidate the two cleanup useEffect hooks (lines 169-183 and 185-203) into a single cleanup effect.

**Current issue:** Two separate useEffect hooks with empty dependency arrays both clean up the same refs (scrollRef, pressTimeout, pressInterval), creating redundant cleanup logic.

**Implementation:**
1. Remove the first cleanup useEffect (lines 169-183)
2. Keep the second cleanup useEffect (lines 185-203) as it already:
   - Sets isMountedRef.current = true on mount
   - Nulls refs after cleanup (prevents double-cleanup)
   - Handles all three refs: scrollRef, pressTimeout, pressInterval
3. Verify all cleanup calls are present:
   - cancelAnimationFrame(scrollRef.current)
   - clearTimeout(pressTimeout.current)
   - clearInterval(pressInterval.current)
4. Add comment documenting consolidated cleanup pattern per research Pattern 1

**Pattern from research:**
```typescript
useEffect(() => {
  isMountedRef.current = true
  return () => {
    isMountedRef.current = false
    // Cancel animation frame
    if (scrollRef.current) {
      cancelAnimationFrame(scrollRef.current)
      scrollRef.current = null
    }
    // Clear timeout
    if (pressTimeout.current) {
      clearTimeout(pressTimeout.current)
      pressTimeout.current = null
    }
    // Clear interval
    if (pressInterval.current) {
      clearInterval(pressInterval.current)
      pressInterval.current = null
    }
  }
}, [])
```

**Why this fixes Pitfall #1:** Single cleanup hook with ref nulling prevents race conditions from multiple cleanup attempts.

```typescript
import { renderHook, act, waitFor } from '@testing-library/react'
import { usePerformanceControls } from '@/hooks/use-performance-controls'
import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest'

describe('usePerformanceControls - Memory Leak Detection', () => {
  beforeEach(() => {
    vi.useFakeTimers()
  })

  afterEach(() => {
    vi.restoreAllMocks()
    vi.useRealTimers()
  })

  it('should clean up all timers on unmount after extended session', async () => {
    const { result, unmount } = renderHook(() =>
      usePerformanceControls({ /* mock props */ })
    )

    // Simulate 30-minute session with typical interactions
    // 10 songs × 3 minutes each = 30 minutes
    // 20 play/pause toggles
    // 15 BPM changes

    for (let i = 0; i < 20; i++) {
      act(() => {
        result.current.handleTogglePlay()
      })
      // Advance timer to simulate 90 seconds between toggles
      act(() => {
        vi.advanceTimersByTime(90000)
      })
    }

    // Verify no timers leaked before unmount
    const pendingTimersBefore = vi.getTimerCount()

    // Unmount - should clean up all timers
    unmount()

    // Verify all timers cleaned up
    const pendingTimersAfter = vi.getTimerCount()
    expect(pendingTimersAfter).toBe(0)
    expect(pendingTimersAfter).toBeLessThan(pendingTimersBefore)
  })

  it('should null refs after cleanup to prevent double-cleanup', () => {
    const { result, unmount } = renderHook(() =>
      usePerformanceControls({ /* mock props */ })
    )

    // Start auto-scroll (creates interval)
    act(() => {
      result.current.handleTogglePlay()
    })

    // Unmount and verify refs are nulled
    unmount()

    // Cannot directly access refs, but verify no errors on second cleanup attempt
    // (if refs not nulled, double clearInterval could throw)
    expect(() => unmount()).not.toThrow()
  })

  it('should handle rapid mount/unmount cycles without memory leak', () => {
    // Simulate component rapidly mounting and unmounting
    // (e.g., user toggling performance mode on/off quickly)

    for (let i = 0; i < 50; i++) {
      const { unmount } = renderHook(() =>
        usePerformanceControls({ /* mock props */ })
      )
      unmount()
    }

    // Verify no accumulated timers
    expect(vi.getTimerCount()).toBe(0)
  })
})
```
