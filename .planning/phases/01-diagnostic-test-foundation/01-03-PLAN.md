```typescript
import { renderHook } from '@testing-library/react'
import { useContentRenderer } from '@/hooks/use-content-renderer'

describe('useContentRenderer - Chords Content', () => {
  it('should return chords renderType for Chords content with sections', () => {
    const chordsData = [{
      sections: [
        { id: '1', name: 'Verse 1', chords: 'C F G', lyrics: 'Test lyrics' },
        { id: '2', name: 'Chorus', chords: 'F C G', lyrics: 'Chorus lyrics' }
      ]
    }]

    const { result } = renderHook(() => useContentRenderer({
      currentSong: 0,
      currentSongData: {
        id: 'test-1',
        title: 'Test Song',
        artist: 'Test Artist',
        content_type: 'Chords',
        content_data: { sections: chordsData[0].sections }
      },
      sheetUrls: [null],
      sheetMimeTypes: [null],
      lyricsData: [],
      chordsData
    }))

    // Document actual hook output for debugging
    console.log('Hook result:', {
      renderType: result.current.renderType,
      hasContent: result.current.hasContent,
      chordsData: result.current.chordsData
    })

    // Assert expected behavior
    expect(result.current.renderType).toBe('chords')
    expect(result.current.hasContent).toBe(true)
    expect(result.current.chordsData).toBeDefined()

    // Verify sections array passed through correctly
    if (result.current.chordsData) {
      expect(result.current.chordsData).toHaveLength(2)
      expect(result.current.chordsData[0].name).toBe('Verse 1')
    }
  })

  it('should handle Chords content with empty sections', () => {
    const chordsData = [{ sections: [] }]

    const { result } = renderHook(() => useContentRenderer({
      currentSong: 0,
      currentSongData: {
        content_type: 'Chords',
        content_data: { sections: [] }
      },
      sheetUrls: [null],
      sheetMimeTypes: [null],
      lyricsData: [],
      chordsData
    }))

    // Edge case: empty sections should still be Chords type
    expect(result.current.renderType).toBe('chords')
    expect(result.current.hasContent).toBe(false) // Or true? Inspect hook logic
  })
})
```

- Provide all required hook parameters (inspect hook to see full signature)
- Log hook output with console.log to document actual vs expected behavior
- Test both normal case (sections present) and edge case (empty sections)

**Expected outcome:** Test may pass OR fail depending on where bug is:
- If test PASSES: Bug is NOT in hook (renderType correct) → bug is in ContentDisplay component
- If test FAILS: Bug IS in hook (wrong renderType or wrong data structure) → hook logic broken

Either outcome is valuable for root cause analysis.

import { ContentDisplay } from '@/components/performance-mode/content-display'

describe('ContentDisplay - Chords Rendering', () => {

    const renderInfo = {
      renderType: 'chords' as const,
      hasContent: true,
      chordsData: [
        { id: '1', name: 'Verse 1', chords: 'C F G Am', lyrics: 'First verse' },
        { id: '2', name: 'Chorus', chords: 'F C G', lyrics: 'Chorus part' }
      ],
      // Include other required renderInfo fields based on component inspection
    }

    render(
      <ContentDisplay
        renderInfo={renderInfo}
        currentSongData={{
          id: 'test-1',
          title: 'Test Song',
          content_type: 'Chords',
          content_data: { sections: renderInfo.chordsData }
        }}
        currentSong={0}
        zoom={100}
      />
    )

    // Verify sections render in DOM
    expect(screen.getByText('Verse 1')).toBeInTheDocument()
    expect(screen.getByText('Chorus')).toBeInTheDocument()

    // Verify chord progressions render
    expect(screen.getByText(/C F G Am/)).toBeInTheDocument()
    expect(screen.getByText(/F C G/)).toBeInTheDocument()
  })

  it('should handle empty sections array', () => {
    const renderInfo = {
      renderType: 'chords' as const,
      hasContent: false,
      chordsData: [],
    }

    const { container } = render(
      <ContentDisplay
        renderInfo={renderInfo}
        currentSongData={{ content_type: 'Chords' }}
        currentSong={0}
        zoom={100}
      />
    )

    // Should render empty state or nothing (not crash)
    expect(container).toBeInTheDocument()
  })
})
